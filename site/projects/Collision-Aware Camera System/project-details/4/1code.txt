public class CameraFollow : MonoBehaviour
{

    //...

    void CollisionDistanceUpdate() {

        float castDistance = maxDistance + 10f;
        Vector3 sideAngleVector = transform.right * sideAngleFactor;

        Vector3 rayLeftDir = (-transform.forward + sideAngleVector).normalized;
        Vector3 rayRightDir = (-transform.forward + -sideAngleVector).normalized;

        Ray rayRight = new Ray(target.transform.position, rayRightDir);
        Debug.DrawRay(rayRight.origin, rayRight.direction * castDistance);


        Ray rayLeft = new Ray(target.transform.position, rayLeftDir);
        Debug.DrawRay(rayLeft.origin, rayLeft.direction * castDistance);

        RaycastHit hit;


        if ((nextCheck == 0 || nextCheck == 1) && Physics.Raycast(rayRight, out hit, castDistance, cameraObstacles)) {
            float adjustedDst =  hit.distance - collisionCameraPadding;
            currentDst = adjustedDst;
            nextCheck = 1;
        }
        else if ((nextCheck == 0 || nextCheck == 2) && Physics.Raycast(rayLeft, out hit, castDistance, cameraObstacles)) {
            float adjustedDst = hit.distance - collisionCameraPadding;
            currentDst = adjustedDst;
            nextCheck = 2;
        }
        else if(nextCheck == 1) {
            if (!Physics.Raycast(rayLeft, out hit, castDistance, cameraObstacles)) {
                currentDst = maxDistance;
                nextCheck = 0;
            }
        }
        else if(nextCheck == 2) {
            if (!Physics.Raycast(rayRight, out hit, castDistance, cameraObstacles)) {
                currentDst = maxDistance;
                nextCheck = 0;
            }
        }

        currentDst = Mathf.Clamp(currentDst, 1, maxDistance);
        Debug.Log("Camera collision current check: " + nextCheck);
    }
}
